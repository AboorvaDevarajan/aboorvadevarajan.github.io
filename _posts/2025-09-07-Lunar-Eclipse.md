<p class="eclipse-intro"><strong>7 September 2025.</strong> My first attempt at photographing a lunar eclipse. <img class="smiley" src="/assets/img/icons/smile.png" alt=""> I started late, when the Moon was already deep in Earth's shadow, and stayed with it until the bright face had almost returned. Two and a half hours, a red Moon, clouds that kept moving through, and far more photographs than I expected.</p>

## Starting late

The first frames were already crescents. I had missed the beginning, but not the part where the Moon seemed to disappear.

<p class="eclipse-closing">Watching it was different from seeing a finished eclipse photograph. Photographing it meant staying with every change: the first crescent, the red Moon, the clouds, and the light coming back. For a first attempt, that is the part I want to remember. <img class="smiley" src="/assets/img/icons/smile.png" alt=""></p>

<script>
(function () {
  var triggers = Array.prototype.slice.call(document.querySelectorAll('.lb-trigger'));
  if (!triggers.length) return;

  var overlay = document.getElementById('lb-overlay');
  var image = document.getElementById('lb-image');
  var caption = document.getElementById('lb-caption');
  var index = 0;

  function openAt(i) {
    index = (i + triggers.length) % triggers.length;
    var link = triggers[index];
    image.src = link.getAttribute('href');
    image.alt = (link.querySelector('img') && link.querySelector('img').alt) || '';
    caption.textContent = link.getAttribute('data-caption') || '';
    overlay.classList.add('is-open');
    overlay.setAttribute('aria-hidden', 'false');
    document.body.style.overflow = 'hidden';
  }

  function closeLb() {
    overlay.classList.remove('is-open');
    overlay.setAttribute('aria-hidden', 'true');
    image.src = '';
    document.body.style.overflow = '';
  }

  triggers.forEach(function (link, i) {
    link.addEventListener('click', function (event) {
      event.preventDefault();
      openAt(i);
    });
  });

  document.getElementById('lb-close').addEventListener('click', closeLb);
  document.getElementById('lb-prev').addEventListener('click', function (event) {
    event.stopPropagation();
    openAt(index - 1);
  });
  document.getElementById('lb-next').addEventListener('click', function (event) {
    event.stopPropagation();
    openAt(index + 1);
  });
  overlay.addEventListener('click', function (event) {
    if (event.target === overlay) closeLb();
  });
  document.addEventListener('keydown', function (event) {
    if (!overlay.classList.contains('is-open')) return;
    if (event.key === 'Escape') closeLb();
    if (event.key === 'ArrowLeft') openAt(index - 1);
    if (event.key === 'ArrowRight') openAt(index + 1);
  });
})();
</script>
